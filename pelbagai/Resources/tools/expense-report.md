Use this tool after receipt scans when the user wants summaries, spending analysis, totals, merchant breakdowns, or charts.

## Fields

| Field | Type | Required |
|-------|------|----------|
| Title | text | no |
| Summary | text | no |
| Currency | text | no |
| Receipt Count | text | no |
| Total Spend | currency | no |
| Average Spend | currency | no |
| Top Merchant | text | no |
| Merchant Spend | list | no |
| Daily Spend | list | no |
| Chart Spec | text | no |

## Rules

- Read saved receipt results instead of inventing transactions.
- Summarize deterministic totals generated by the app workflow.
- Use the Chart Spec field for charts when rendering a report.

## Prompt

Summarize saved receipt data into a concise expense report. Do not invent receipt rows.

## Workflow

```yaml
triggerSourceToolIDs: [receipt]
triggerKeywords: [report, summary, summarize, summarise, chart, spending, expense, analyse, analyze]
inputToolIDs: [receipt]
saveLatestSourceResult: true
```

```javascript
var receipts = (input.recordsByTool.receipt || []);

function field(row, names) {
  var fields = row.richFields || {};
  for (var i = 0; i < names.length; i++) {
    var value = fields[names[i]];
    if (Array.isArray(value)) { value = value.join(', '); }
    if (value !== undefined && value !== null && String(value).trim() !== '') {
      return String(value).trim();
    }
  }
  return '';
}

function amount(value) {
  var match = String(value || '').replace(/,/g, '').match(/[-+]?\d*\.?\d+/g);
  return match && match.length ? Number(match[match.length - 1]) : NaN;
}

function money(value, currency) {
  var amountText = (Math.round(value * 100) / 100).toFixed(2);
  return currency ? currency + ' ' + amountText : amountText;
}

function dateBucket(value) {
  value = String(value || '').trim();
  return value || 'Undated';
}

var rows = receipts.map(function(row) {
  var total = amount(field(row, ['Total', 'Amount', 'Grand Total', 'Final Amount', 'total', 'amount']));
  if (!isFinite(total)) { return null; }
  return {
    id: row.id,
    merchant: field(row, ['Store Name', 'Merchant', 'Shop', 'Vendor', 'Business', 'store_name']) || 'Unknown Merchant',
    currency: field(row, ['Currency', 'currency']).toUpperCase(),
    total: total,
    date: dateBucket(field(row, ['Date', 'date', 'Transaction Date']))
  };
}).filter(Boolean);

var total = rows.reduce(function(sum, row) { return sum + row.total; }, 0);
var average = rows.length ? total / rows.length : 0;

var currencyCounts = {};
rows.forEach(function(row) {
  if (row.currency) {
    currencyCounts[row.currency] = (currencyCounts[row.currency] || 0) + 1;
  }
});
var currency = Object.keys(currencyCounts).sort(function(a, b) {
  return currencyCounts[b] - currencyCounts[a];
})[0] || '';

var merchantTotals = {};
var dailyTotals = {};
rows.forEach(function(row) {
  merchantTotals[row.merchant] = (merchantTotals[row.merchant] || 0) + row.total;
  dailyTotals[row.date] = (dailyTotals[row.date] || 0) + row.total;
});
var merchants = Object.keys(merchantTotals).sort(function(a, b) {
  return merchantTotals[b] - merchantTotals[a];
});
var days = Object.keys(dailyTotals).sort();
var topMerchant = merchants[0] || '';

var summary = rows.length
  ? ('Summarized ' + rows.length + ' receipt' + (rows.length === 1 ? '' : 's')
    + ' with total spend of ' + money(total, currency)
    + '. Average receipt value is ' + money(average, currency)
    + (topMerchant ? '. Largest merchant by spend is ' + topMerchant
      + ' at ' + money(merchantTotals[topMerchant], currency) + '.' : '.'))
  : 'No saved receipt rows with parseable totals were found.';

return {
  title: 'Expense Report',
  summary: summary,
  metrics: [
    { title: 'Receipts', value: String(rows.length) },
    { title: 'Total', value: money(total, currency) },
    { title: 'Average', value: money(average, currency) }
  ],
  charts: [
    {
      title: 'Merchant spend',
      kind: 'bar',
      points: merchants.slice(0, 6).map(function(name) {
        return { label: name, value: Math.round(merchantTotals[name] * 100) / 100 };
      })
    },
    {
      title: 'Daily spend',
      kind: 'line',
      points: days.map(function(day) {
        return { label: day, value: Math.round(dailyTotals[day] * 100) / 100 };
      })
    }
  ],
  fields: {
    Currency: currency,
    'Receipt Count': String(rows.length),
    'Total Spend': money(total, currency),
    'Average Spend': money(average, currency),
    'Top Merchant': topMerchant
  },
  sourceResultIDs: rows.map(function(row) { return row.id; })
};
```
